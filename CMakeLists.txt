cmake_minimum_required(VERSION 3.16)
project(rf_observatory VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rf_core core/src/scanner.cpp)
target_include_directories(rf_core PUBLIC core/include)
target_compile_options(rf_core PRIVATE -Wall -Wextra -Werror -Wconversion)
add_executable(rf_simulator simulator/main.cpp)
target_link_libraries(rf_simulator PRIVATE rf_core)
enable_testing()
add_executable(rf_tests tests/core_test.cpp)
target_link_libraries(rf_tests PRIVATE rf_core)
add_test(NAME rf_core COMMAND rf_tests)
